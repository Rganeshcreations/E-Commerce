import React, { useCallback, useEffect, useState } from "react";
import { api } from "../services/api";

export default function Tenants() {
  const [tenants, setTenants] = useState([]);
  const [loading, setLoading] = useState(true);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");
  const [name, setName] = useState("");
  const [slug, setSlug] = useState("");

  const loadTenants = useCallback(async () => {
    setLoading(true);
    setError("");
    try {
      const data = await api.tenants();
      setTenants(Array.isArray(data) ? data : data?.content ?? []);
    } catch (e) {
      setError(e.message || "Unable to load tenants.");
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { loadTenants(); }, [loadTenants]);

  async function createTenant(e) {
    e.preventDefault();
    setBusy(true);
    setError("");
    setMessage("");
    try {
      await api.createTenant({ name: name.trim(), slug: slug.trim().toLowerCase() });
      setName("");
      setSlug("");
      setMessage("Tenant created successfully.");
      await loadTenants();
    } catch (e) {
      setError(e.message || "Unable to create tenant.");
    } finally {
      setBusy(false);
    }
  }

  async function toggleTenant(tenant) {
    setBusy(true);
    setError("");
    setMessage("");
    try {
      await api.setTenantActive(tenant.id, !tenant.active);
      setMessage(`Tenant ${tenant.active ? "deactivated" : "activated"} successfully.`);
      await loadTenants();
    } catch (e) {
      setError(e.message || "Unable to update tenant status.");
    } finally {
      setBusy(false);
    }
  }

  const styles = {
    page: { maxWidth: 1100, margin: "0 auto", padding: 28, color: "#17233c" },
    card: { background: "#fff", border: "1px solid #e2e7ef", borderRadius: 14, padding: 20, marginBottom: 22 },
    field: { display: "block", width: "100%", boxSizing: "border-box", padding: 11, border: "1px solid #dce3ec", borderRadius: 8, margin: "7px 0 14px" },
    button: { border: 0, borderRadius: 8, padding: "10px 15px", background: "#2864e6", color: "#fff", cursor: "pointer", fontWeight: 700 },
    table: { width: "100%", borderCollapse: "collapse" },
    cell: { padding: 13, borderBottom: "1px solid #edf0f5", textAlign: "left" }
  };

  return (
    <main style={styles.page}>
      <h1>Tenant Management</h1>
      <p>Manage marketplace stores and their active status.</p>

      <section style={styles.card}>
        <h2>Create tenant</h2>
        <form onSubmit={createTenant}>
          <label>Store name
            <input style={styles.field} value={name} onChange={e => setName(e.target.value)} required maxLength={100} />
          </label>
          <label>Store slug
            <input style={styles.field} value={slug} onChange={e => setSlug(e.target.value)} required pattern="[a-z0-9]+(-[a-z0-9]+)*" title="Use lowercase letters, numbers and hyphens." />
          </label>
          <button style={styles.button} disabled={busy} type="submit">
            {busy ? "Please wait..." : "Create tenant"}
          </button>
        </form>
      </section>

      {error && <p role="alert" style={{ color: "#b42318" }}>{error}</p>}
      {message && <p role="status" style={{ color: "#17814b" }}>{message}</p>}

      <section style={styles.card}>
        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", gap: 12 }}>
          <h2>Registered tenants</h2>
          <button style={styles.button} onClick={loadTenants} disabled={loading || busy}>Refresh</button>
        </div>

        {loading ? <p>Loading tenants...</p> : tenants.length === 0 ? <p>No tenants found.</p> : (
          <div style={{ overflowX: "auto" }}>
            <table style={styles.table}>
              <thead><tr>{["ID", "Store", "Slug", "Status", "Created", "Action"].map(h => <th key={h} style={styles.cell}>{h}</th>)}</tr></thead>
              <tbody>
                {tenants.map(t => (
                  <tr key={t.id}>
                    <td style={styles.cell}>{t.id}</td>
                    <td style={styles.cell}>{t.name}</td>
                    <td style={styles.cell}>{t.slug}</td>
                    <td style={styles.cell}>{t.active ? "Active" : "Inactive"}</td>
                    <td style={styles.cell}>{t.createdAt ? new Date(t.createdAt).toLocaleString() : "—"}</td>
                    <td style={styles.cell}>
                      <button style={styles.button} disabled={busy} onClick={() => toggleTenant(t)}>
                        {t.active ? "Deactivate" : "Activate"}
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </section>
    </main>
  );
}

const BASE = "http://localhost:8080";

async function request(path, options = {}) {

  const token =
    localStorage.getItem("marketgrid_token");

  const headers = {
    "Content-Type": "application/json",
    ...(options.headers || {})
  };

  if (token) {
    headers.Authorization = `Bearer ${token}`;
  }

  const response = await fetch(
    `${BASE}${path}`,
    {
      ...options,
      headers
    }
  );

  const text = await response.text();

  let data = null;

  try {
    data = text ? JSON.parse(text) : null;
  } catch {
    data = text;
  }

  if (!response.ok) {

    throw new Error(
      data?.message ||
      data?.error ||
      `Request failed (${response.status})`
    );
  }

  return data;
}
